################################################################################
# Target-specific paths for multi-binary compilation                           #
################################################################################

set(CHIMERA_TARGET_DIR
    ${CMAKE_CURRENT_LIST_DIR}
    CACHE PATH "Directory containing target-specific files for Chimera Gen" FORCE
)

################################################################################
# Dynamically generated resources (single source of truth)                     #
################################################################################
# chimera-gen is functionally identical to chimera-open, except it pulls as many
# resources as possible from the artifacts generated by the chimera-top RTL repo
# instead of shipping committed copies:
#
#   * RDL    (SystemRDL -> peakrdl, `make rdl`): SoC address map + register bases
#            in chimera-top/.generated/ and the linker address header
#            sw/link/chimera_addrs.ldh.
#   * reggen (lowRISC regtool, `make otp`): OpenTitan peripheral register headers
#            (gpio, i2c, spi_host, rv_plic) in opentitan_peripherals/sw/include/.
#
# Register blocks whose generated headers are not yet API-compatible with the SDK
# drivers (cheshire/clint/hyperbus/soc_ctrl, snitch_cluster_peripheral) are still
# carried as committed copies here, exactly as in chimera-open. See
# chimera-top/docs/memory-map.md for the migration plan.

# This SDK lives at chimera-top/sw/deps/chimera-sdk; PROJECT_SOURCE_DIR is the
# SDK root, so chimera-top is three levels up.
get_filename_component(CHIMERA_TOP_ROOT "${PROJECT_SOURCE_DIR}/../../.." ABSOLUTE)

set(CHIMERA_RDL_GEN_DIR "${CHIMERA_TOP_ROOT}/.generated"
    CACHE PATH "Directory holding the peakrdl-generated SoC headers")
if(NOT EXISTS "${CHIMERA_RDL_GEN_DIR}/chimera_addrmap_raw.h")
    message(FATAL_ERROR
        "[CHIMERA-SDK] chimera-gen: RDL-generated headers not found in "
        "'${CHIMERA_RDL_GEN_DIR}'. Run `make rdl` in chimera-top first "
        "(overridable via -D CHIMERA_RDL_GEN_DIR=...).")
endif()

set(CHIMERA_REGGEN_INC_DIR
    "${PROJECT_SOURCE_DIR}/host/third_party/opentitan/opentitan_peripherals/sw/include"
    CACHE PATH "Directory holding the reggen-generated OT peripheral headers")
if(NOT EXISTS "${CHIMERA_REGGEN_INC_DIR}/gpio_regs.h")
    message(FATAL_ERROR
        "[CHIMERA-SDK] chimera-gen: reggen-generated peripheral headers not "
        "found in '${CHIMERA_REGGEN_INC_DIR}'. Run `make otp` in the "
        "opentitan_peripherals dependency first "
        "(overridable via -D CHIMERA_REGGEN_INC_DIR=...).")
endif()

# Linker address map (sw/link/chimera_addrs.ldh, `make rdl`). shared/common.ldh
# pulls it in via `INCLUDE link/chimera_addrs.ldh`. We put chimera-top/sw (the
# PARENT of link/) on the search path — not sw/link itself — so the bare
# `INCLUDE common.ldh` in the SDK link templates still resolves to this target's
# shared/common.ldh rather than chimera-top's own sw/link/common.ldh. Set on the
# SDK top-level directory so the test binaries (added after targets/) inherit -L.
set(CHIMERA_LINK_DIR "${CHIMERA_TOP_ROOT}/sw/link"
    CACHE PATH "Directory holding the peakrdl-generated linker address header")
if(NOT EXISTS "${CHIMERA_LINK_DIR}/chimera_addrs.ldh")
    message(FATAL_ERROR
        "[CHIMERA-SDK] chimera-gen: generated linker address header not found "
        "in '${CHIMERA_LINK_DIR}'. Run `make rdl` in chimera-top first "
        "(overridable via -D CHIMERA_LINK_DIR=...).")
endif()
set_property(DIRECTORY "${PROJECT_SOURCE_DIR}" APPEND
    PROPERTY LINK_OPTIONS "-L${CHIMERA_TOP_ROOT}/sw")

message(STATUS "[CHIMERA-SDK] chimera-gen RDL headers    : ${CHIMERA_RDL_GEN_DIR}")
message(STATUS "[CHIMERA-SDK] chimera-gen reggen headers : ${CHIMERA_REGGEN_INC_DIR}")
message(STATUS "[CHIMERA-SDK] chimera-gen linker header  : ${CHIMERA_LINK_DIR}/chimera_addrs.ldh")

################################################################################
# Host Runtime Configuration                                                    #
################################################################################
set_property(GLOBAL PROPERTY CHIMERA_HOST_RUNTIME_SOURCES
    ${CMAKE_CURRENT_LIST_DIR}/host/src/crt0_64.S
    ${CMAKE_CURRENT_LIST_DIR}/shared/src/cluster_addr_map.c
    ${CMAKE_CURRENT_LIST_DIR}/shared/src/shared.c
)
set_property(GLOBAL PROPERTY CHIMERA_HOST_RUNTIME_INCLUDES
    ${CMAKE_CURRENT_LIST_DIR}/shared/inc
    ${CMAKE_CURRENT_LIST_DIR}/shared/inc/regs
    ${CMAKE_CURRENT_LIST_DIR}/shared/inc/addr_maps
    # Dynamically generated headers resolved above
    ${CHIMERA_RDL_GEN_DIR}
    ${CHIMERA_REGGEN_INC_DIR}
    ${PROJECT_SOURCE_DIR}/host/hal/inc
)
set_property(GLOBAL PROPERTY CHIMERA_HOST_RUNTIME_DEFINES
    TARGET_PLATFORM_CHIMERA_GEN
    OT_PLATFORM_RV32
)
set_property(GLOBAL PROPERTY CHIMERA_HOST_RUNTIME_PRIVATE_INCLUDES "")
